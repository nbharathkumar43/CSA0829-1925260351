import random
import time

# Generate rainfall readings for 365 days
readings = [round(random.uniform(0, 30), 2) for _ in range(365)]

# Weekly rainfall totals
weekly_totals = []

for i in range(0, len(readings), 7):
    week = readings[i:i + 7]
    weekly_totals.append(round(sum(week), 2))

# Wettest and driest weeks
wettest_week = weekly_totals.index(max(weekly_totals)) + 1
driest_week = weekly_totals.index(min(weekly_totals)) + 1

# Naive moving average
def naive_average(data):
    averages = []

    for i in range(len(data) - 6):
        total = sum(data[i:i + 7])
        averages.append(total / 7)

    return averages

# Running-sum moving average
def running_average(data):
    averages = []

    if len(data) < 7:
        return averages

    total = sum(data[:7])
    averages.append(total / 7)

    for i in range(7, len(data)):
        total = total - data[i - 7] + data[i]
        averages.append(total / 7)

    return averages

# Dry days
dry_days = [i + 1 for i, value in enumerate(readings) if value == 0]

print("Total readings:", len(readings))
print("Number of weeks:", len(weekly_totals))
print("Wettest week:", wettest_week)
print("Driest week:", driest_week)
print("First 5 weekly totals:", weekly_totals[:5])
print("Number of moving averages:", len(running_average(readings)))
print("Dry days:", dry_days)

# Experimental investigation
for size in [365, 36500, 365000]:
    data = [random.uniform(0, 30) for _ in range(size)]

    start = time.perf_counter()
    naive_average(data)
    naive_time = time.perf_counter() - start

    start = time.perf_counter()
    running_average(data)
    running_time = time.perf_counter() - start

    print("\nReadings:", size)
    print("Naive time:", naive_time)
    print("Running-sum time:", running_time)

# Edge cases
print("\nEmpty list:", running_average([]))
print("Short list:", running_average([2, 3, 4]))

import random
import string
import time
from collections import Counter

calls = [
    "seeds not received seeds delayed",
    "fertiliser price high",
    "seeds price high subsidy not received"
]

STOP = {"not", "high"}

# Count word frequencies
def frequencies(texts, stop):
    counts = {}

    for text in texts:
        words = text.lower().translate(
            str.maketrans("", "", string.punctuation)
        ).split()

        for word in words:
            if word not in stop:
                counts[word] = counts.get(word, 0) + 1

    return counts

# Find top k words
def top(counts, k):
    return sorted(
        counts.items(),
        key=lambda item: (-item[1], item[0])
    )[:k]

print("Top words:", top(frequencies(calls, STOP), 3))

# Group complaints by district
complaints = [
    ("Chennai", "Seeds not received"),
    ("Madurai", "Water supply problem"),
    ("Chennai", "Fertiliser price high"),
    ("Salem", "Crop damage")
]

district_groups = {}

for district, complaint in complaints:
    district_groups.setdefault(district, []).append(complaint)

print("\nComplaints by district:")

for district, items in district_groups.items():
    print(district, ":", items)

# Experimental investigation
sample_words = ["seeds", "price", "water", "seeds", "crop"]

print("\nCounter result:", Counter(sample_words))

# Compare dictionary counting and list.count()
for size in [1000, 10000, 100000]:
    data = [random.choice(sample_words) for _ in range(size)]

    start = time.perf_counter()
    result = {}

    for word in data:
        result[word] = result.get(word, 0) + 1

    dictionary_time = time.perf_counter() - start

    start = time.perf_counter()
    distinct = set(data)
    list_result = {word: data.count(word) for word in distinct}
    list_count_time = time.perf_counter() - start

    start = time.perf_counter()
    counter_result = Counter(data)
    counter_time = time.perf_counter() - start

    print("\nComplaints:", size)
    print("Dictionary time:", dictionary_time)
    print("List.count time:", list_count_time)
    print("Counter time:", counter_time)

# Edge cases

print("Upper/lower case:", frequencies(
    ["SEEDS seeds Seeds"], STOP
))
print("Punctuation:", frequencies(
    ["Seeds, seeds! price."], STOP
))

import re
import time
import random

# Normalise phone number
def normalise_number(number):
    if not number:
        return ""

    digits = "".join(
        character for character in number
        if character.isdigit()
    )

    if len(digits) == 12 and digits.startswith("91"):
        digits = digits[2:]

    return digits

# Normalise name
def normalise_name(name):
    return " ".join(name.strip().lower().split())

# Merge contacts
def merge_contacts(contacts):
    merged = {}
    missing_phone = []

    for contact in contacts:
        name = contact.get("name", "").strip()
        phone = normalise_number(contact.get("phone", ""))
        email = contact.get("email", "").strip().lower()

        if not phone:
            missing_phone.append({
                "name": name,
                "email": email
            })
            continue

        if phone not in merged:
            merged[phone] = {
                "names": set(),
                "phones": set(),
                "emails": set()
            }

        if name:
            merged[phone]["names"].add(name)

        merged[phone]["phones"].add(phone)

        if email:
            merged[phone]["emails"].add(email)

    return merged, missing_phone

# Sample contacts
